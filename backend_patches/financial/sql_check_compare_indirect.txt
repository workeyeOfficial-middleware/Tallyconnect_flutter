grp AS (
  SELECT name FROM tally_groups
   WHERE admin_id = 3 AND company_guid = 'f868f002-3e6c-4f78-b558-5bfdadd0cceb'
     AND name IN ('Indirect Expenses', 'Indirect Incomes')
  UNION
  SELECT g.name FROM tally_groups g JOIN grp ON g.parent = grp.name
   WHERE g.admin_id = 3 AND g.company_guid = 'f868f002-3e6c-4f78-b558-5bfdadd0cceb'

mv AS (
  SELECT lower(btrim(regexp_replace(ve.ledger_name, '[[:space:]]+', ' ', 'g'))) AS k,
         SUM(CASE WHEN ve.is_debit THEN -ve.amount ELSE ve.amount END) AS app_dr,
         COUNT(DISTINCT ve.voucher_guid) AS vouchers
    FROM voucher_entries ve
   WHERE ve.admin_id = 3 AND ve.company_guid = 'f868f002-3e6c-4f78-b558-5bfdadd0cceb'
     AND COALESCE(ve.is_active, TRUE) = TRUE
     AND ve.voucher_date BETWEEN '2016-04-01' AND '2026-10-21'
     AND COALESCE(ve.voucher_type, '') !~* '(memorandum|reversing journal|optional)'

  SELECT l.name, l.parent_group, COALESCE(l.opening_balance, 0) AS opening,
         lower(btrim(regexp_replace(l.name, '[[:space:]]+', ' ', 'g'))) AS k
    FROM ledgers l
   WHERE l.admin_id = 3 AND l.company_guid = 'f868f002-3e6c-4f78-b558-5bfdadd0cceb'
)
SELECT COALESCE(t.name, app.name) AS ledger,
       t.tally_group,
       app.parent_group AS app_group,
       t.tally_dr,
       COALESCE(mv.app_dr, 0) AS app_dr,
       COALESCE(mv.app_dr, 0) - COALESCE(t.tally_dr, 0) AS diff,
       app.opening,
       mv.vouchers
  FROM t
  FULL JOIN app ON app.k = lower(btrim(regexp_replace(t.name, '[[:space:]]+', ' ', 'g')))
  LEFT JOIN mv ON mv.k = COALESCE(app.k, lower(btrim(regexp_replace(t.name, '[[:space:]]+', ' ', 'g'))))

   AND ABS(COALESCE(mv.app_dr, 0) - COALESCE(t.tally_dr, 0)) > 0.01
 ORDER BY ABS(COALESCE(mv.app_dr, 0) - COALESCE(t.tally_dr, 0)) DESC;
